# An optional build path for hosts without BSD make.  The BSD Makefile is
# still the canonical build on FreeBSD; keep the two source lists in sync.
cmake_minimum_required(VERSION 3.16)
project(ucas C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# Every src/*.c is part of the program.  The BSD Makefile lists the same
# files explicitly; a GLOB keeps this path from falling behind a new source
# (CONFIGURE_DEPENDS re-globs when one is added).
file(GLOB UCAS_SOURCES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/cfx86.c")
  set(UCAS_HAVE_X86 TRUE)
endif()

# The GLSL ES compiler is kept as its own library: its glslc.c would clash
# with the top-level glslc.c as an object.
set(UCAS_SNX_SOURCES
  src/snxglslc/lexer.c src/snxglslc/parser.c src/snxglslc/emit.c
  src/snxglslc/builtins.c src/snxglslc/glslc.c
)
add_library(ucas_snxglslc STATIC ${UCAS_SNX_SOURCES})
set_target_properties(ucas_snxglslc PROPERTIES
  POSITION_INDEPENDENT_CODE ON
  C_STANDARD 99
  C_STANDARD_REQUIRED ON
  C_EXTENSIONS OFF)
target_include_directories(ucas_snxglslc PRIVATE src/snxglslc)

# Two headers are generated by the BSD Makefile from source text.
set(UCAS_NLP_WORDS "${CMAKE_CURRENT_BINARY_DIR}/nlp_words.h")
set(UCAS_THY "${CMAKE_CURRENT_BINARY_DIR}/thy_absalg.inc")
add_custom_command(
  OUTPUT "${UCAS_NLP_WORDS}"
  COMMAND sh -c "awk -f '${CMAKE_CURRENT_SOURCE_DIR}/src/nlp_words.awk' \
    '${CMAKE_CURRENT_SOURCE_DIR}/src/nlp.c' \
    '${CMAKE_CURRENT_SOURCE_DIR}/src/nlp_calc2.inc' \
    '${CMAKE_CURRENT_SOURCE_DIR}/src/nlp_hyper.inc' \
    '${CMAKE_CURRENT_SOURCE_DIR}/src/nlp_alg.inc' \
    '${CMAKE_CURRENT_SOURCE_DIR}/src/nlp_xform.inc' \
    '${CMAKE_CURRENT_SOURCE_DIR}/src/nlp_img.inc' > '${UCAS_NLP_WORDS}'"
  DEPENDS
    src/nlp.c src/nlp_calc2.inc src/nlp_hyper.inc src/nlp_alg.inc
    src/nlp_xform.inc src/nlp_img.inc src/nlp_words.awk
  VERBATIM)
add_custom_command(
  OUTPUT "${UCAS_THY}"
  COMMAND sh -c "sed -e 's/\\\\/\\\\\\\\/g' -e 's/\"/\\\\\"/g' \
    -e 's/^/\"/' -e 's/\$/\\\\n\"/' \
    '${CMAKE_CURRENT_SOURCE_DIR}/src/thy_absalg.thy' > '${UCAS_THY}'"
  DEPENDS src/thy_absalg.thy
  VERBATIM)

add_executable(ucas ${UCAS_SOURCES} "${UCAS_NLP_WORDS}" "${UCAS_THY}")
add_dependencies(ucas ucas_snxglslc)
target_include_directories(ucas PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}/src"
  "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ucas PRIVATE ucas_snxglslc m)

if(UCAS_HAVE_X86)
  target_compile_definitions(ucas PRIVATE CF_HAVE_X86)
endif()

# strdup and strcasecmp are POSIX, not ISO C99.  Keep _GNU_SOURCE off: on
# glibc it declares fdiv(), which used to clash with find.c's helper.
target_compile_definitions(ucas PRIVATE
  _POSIX_C_SOURCE=200809L
  _DEFAULT_SOURCE)

target_compile_options(ucas PRIVATE
  -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare
  -Wno-missing-field-initializers)

# stb_image is vendored and unmodified; its warnings are not ours.
set_source_files_properties(src/img.c PROPERTIES COMPILE_OPTIONS "-w")

enable_testing()
add_test(NAME sat-selftest COMMAND ucas --sat-selftest)
add_test(NAME portable-smoke
  COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/tools/smoke.sh" "$<TARGET_FILE:ucas>")
set_tests_properties(portable-smoke PROPERTIES
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  ENVIRONMENT "UCAS_SPIRV_CROSS=none")
